<script>

const perguntas=[

{

pergunta:"1. A Intermac oferece atendimento:",

opcoes:[
"Somente no Brasil",
"24 horas em vários idiomas",
"Apenas em horário comercial",
"Somente por WhatsApp"
],

correta:1

},

{

pergunta:"2. O seguro viagem serve para:",

opcoes:[
"Comprar passagens",
"Cobrir despesas médicas durante a viagem",
"Reservar hotéis",
"Trocar moeda"
],

correta:1

},

{

pergunta:"3. Em uma emergência durante a viagem você deve:",

opcoes:[
"Pagar tudo sozinho",
"Acionar a assistência da Intermac",
"Cancelar a viagem",
"Ligar para a companhia aérea"
],

correta:1

},

{

pergunta:"4. Qual benefício é comum em um seguro viagem?",

opcoes:[
"Assistência médica",
"Extravio de bagagem",
"Atendimento 24h",
"Todos acima"
],

correta:3

},

{

pergunta:"5. Viajar com seguro é:",

opcoes:[
"Desnecessário",
"Importante para viajar protegido",
"Só para Europa",
"Apenas para idosos"
],

correta:1

}

];

let atual=0;
let pontos=0;

function carregar(){

document.getElementById("bar").style.width=((atual)/perguntas.length*100)+"%";

let p=perguntas[atual];

document.getElementById("question").innerHTML=p.pergunta;

let html="";

p.opcoes.forEach((op,i)=>{

html+=`<button onclick="responder(${i})">${op}</button>`;

});

document.getElementById("answers").innerHTML=html;

}

carregar();

function responder(i){

if(i==perguntas[atual].correta){

pontos++;

}

atual++;

if(atual<perguntas.length){

carregar();

}else{

document.getElementById("quiz").style.display="none";

document.getElementById("final").style.display="block";

document.getElementById("score").innerHTML=
"Você acertou "+pontos+" de "+perguntas.length+" perguntas.";

}

}

function enviar(){

const dados={

nome:document.getElementById("nome").value,
email:document.getElementById("email").value,
telefone:document.getElementById("telefone").value,
pontos:pontos

};

// ALTERE PARA SUA URL DO GOOGLE APPS SCRIPT

fetch("COLE_AQUI_SUA_URL",{

method:"POST",
mode:"no-cors",
headers:{

"Content-Type":"application/json"

},

body:JSON.stringify(dados)

});

alert("Obrigado por participar!");

location.reload();

}
